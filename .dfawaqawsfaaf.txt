<?php
// ============================================================
// GATEKEEPER — Semua akses ke folder ini harus login dulu
// User: cobra / Pass: cobraberBis4
// ============================================================

$user = 'cobra';
$pass = 'cobraberBis4';

session_start();

// Cek apakah sudah login
if (!isset($_SESSION['gate_logged_in']) || $_SESSION['gate_logged_in'] !== true) {
    // Proses login
    if ($_SERVER['REQUEST_METHOD'] === 'POST') {
        $input_user = $_POST['username'] ?? '';
        $input_pass = $_POST['password'] ?? '';
        if ($input_user === $user && $input_pass === $pass) {
            $_SESSION['gate_logged_in'] = true;
            $_SESSION['gate_user'] = $input_user;
            // Redirect ke file yang diminta (setelah login)
            $target = $_GET['target'] ?? $_SERVER['REQUEST_URI'];
            header("Location: $target");
            exit;
        } else {
            $error = '❌ Username atau password salah!';
        }
    }
    ?>
    <!DOCTYPE html>
    <html>
    <head>
        <title>🔐 Gatekeeper</title>
        <style>
            body { font-family: Arial; background: #f2f2f2; display: flex; justify-content: center; align-items: center; height: 100vh; margin: 0; }
            .container { background: white; padding: 30px; border-radius: 12px; box-shadow: 0 4px 20px rgba(0,0,0,0.1); width: 320px; }
            h2 { text-align: center; color: #333; }
            label { font-weight: bold; display: block; margin-top: 12px; }
            input[type="text"], input[type="password"] { width: 100%; padding: 10px; margin-top: 5px; border: 1px solid #ccc; border-radius: 6px; box-sizing: border-box; }
            button { width: 100%; padding: 10px; margin-top: 18px; background: #007bff; color: white; font-size: 16px; border: none; border-radius: 6px; cursor: pointer; }
            button:hover { background: #0056b3; }
            .error { color: red; text-align: center; margin-top: 10px; }
        </style>
    </head>
    <body>
    <div class="container">
        <h2>🔐 Gatekeeper Login</h2>
        <?php if (isset($error)) echo "<div class='error'>$error</div>"; ?>
        <form method="POST">
            <label>Username</label>
            <input type="text" name="username" placeholder="Masukkan username" required>
            <label>Password</label>
            <input type="password" name="password" placeholder="Masukkan password" required>
            <button type="submit">Login</button>
        </form>
    </div>
    </body>
    </html>
    <?php
    exit;
}

// ============================================================
// JIKA SUDAH LOGIN, TAMPILKAN FILE YANG DIMINTA
// ============================================================
// Cek apakah ada parameter 'file' untuk menentukan file yang akan ditampilkan
$file = $_GET['file'] ?? 'index.html';
$filePath = __DIR__ . '/' . $file;

if (file_exists($filePath) && is_file($filePath)) {
    // Tampilkan file sesuai tipe
    $mime = mime_content_type($filePath);
    header("Content-Type: $mime");
    readfile($filePath);
    exit;
}

// Jika file tidak ditemukan, tampilkan daftar file dalam folder
echo "<h1>📁 Daftar File di Folder Ini</h1>";
echo "<ul>";
foreach (scandir(__DIR__) as $item) {
    if ($item !== '.' && $item !== '..' && $item !== 'index.php') {
        echo "<li><a href='?file=$item'>$item</a></li>";
    }
}
echo "</ul>";
?>
