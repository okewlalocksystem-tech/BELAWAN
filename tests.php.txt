<?php
// Cek syntax seluruh file
$files = ['index.php']; // Ganti dengan nama file utama Anda
foreach ($files as $file) {
    echo "Checking $file... ";
    $output = shell_exec("php -l $file 2>&1");
    if (strpos($output, 'No syntax errors') !== false) {
        echo "✅ OK<br>";
    } else {
        echo "❌ ERROR:<br><pre>$output</pre>";
    }
}
?>
